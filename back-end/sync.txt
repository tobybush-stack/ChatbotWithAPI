# code in main
async def run_sync(provider_id: int):
    providers[provider_id].status = "Syncing..."
    await asyncio.sleep(5)
    providers[provider_id].status = "Complete"

@app.post("/sync/{provider_id}")
async def start_sync(provider_id: int, background_tasks: BackgroundTasks):
    background_tasks.add_task(run_sync, provider_id)
    return {"message": "Sync started"}
